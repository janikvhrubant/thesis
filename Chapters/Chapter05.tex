\chapter{Theoretical Foundations of Deep Neural Networks}
\label{chapter:dnn-theory}
% ------------------------------------------------------------------------------

\acfp{dnn} have become a powerful class of surrogate models in scientific
computing, particularly for high-dimensional and expensive-to-evaluate
functions. This chapter introduces the mathematical structure and training
procedure of \acp{dnn}, with a focus on activation functions, generalization
theory and gradient-based optimization algorithms. It follows the framework
established in \cite{mishra2021enhancing}.

% ------------------------------------------------------------------------------
\section{Theoretical Foundations of Deep Neural Networks}
\label{sec:dnn-definition}
% ------------------------------------------------------------------------------

The key feature enabling neural networks to approximate nonlinear functions is
the application of scalar activation functions. We begin by introducing these
components before defining the class of \acl{dnn} functions.

\begin{definition}[Activation Function] \ \\
An \emph{activation function} is a measurable function $\sigma : \mathbb{R} \to
\mathbb{R}$, typically nonlinear, that is applied componentwise to the output of
each layer in a neural network.
\end{definition}

\begin{remark} \ \\
    In general the activation function $\sigma$ is 1-dimensional, i.e. $\sigma: \mathbb{R} \to \mathbb{R}$. However, it is common to apply it componentwise to vectors and extend the definition to $\sigma: \mathbb{R}^k \to \mathbb{R}^k$ with $\sigma(z) = (\sigma(z_1), \dots, \sigma(z_k))^\top$.
\end{remark}

\begin{example}[Common Activation Functions] \ \\
Two widely used activation functions are given below and illustrated in
Figure~\ref{fig:activation_functions}. These are also the activation functions
we employ in the experimental evaluations presented in Chapter~\ref{chapter7}.

\begin{itemize}
    \item \textbf{Tanh}:
    \begin{equation*}
        \sigma(z) = \tanh(z) = \frac{e^z - e^{-z}}{e^z + e^{-z}}
    \end{equation*}
    The hyperbolic tangent function maps real numbers to $(-1,1)$ and is
    differentiable everywhere. It is often used in hidden layers of neural
    networks.
    
    \item \textbf{Sigmoid}:
    \begin{equation*}
        \sigma(z) = \frac{1}{1 + e^{-z}}
    \end{equation*}
    The sigmoid function maps real numbers to $(0,1)$ and is differentiable
    everywhere. It is often used in theoretical analysis and binary
    classification tasks.
\end{itemize}
\end{example}

  \centering

  \caption{Comparison of ReLU and Sigmoid activation functions.}
  \label{fig:activation_functions}
\end{figure}

With an activation function in place, we now define the term \acl{dnn}
by introducing the representing function that a fully connected feedforward neural network is expressing\cite{mishra2021enhancing}.

\begin{definition}[Fully Connected Deep Neural Network] \ \\
Let $L \in \mathbb{N}$ and $\mathbf{d} = (d_0, d_1, \dots, d_L)$ be a sequence
of positive integers, where $d_0 = s$ is the input dimension and $d_L = 1$ the
output dimension. A \acf{dnn} of \emph{depth} $L$ and \emph{layer widths} $d_1,
\dots, d_{L-1}$ is a function $\hat{f}_\theta: \mathbb{R}^s \to \mathbb{R}$ of the
form
\begin{equation}
    \label{eq:dnn-definition}
    \hat{f}_\theta(y) = A_L \circ \sigma \circ A_{L-1} \circ \cdots \circ \sigma \circ A_1(y),
\end{equation}
where each \emph{layer map} $A_l: \mathbb{R}^{d_{l-1}} \to \mathbb{R}^{d_l}$ is
an affine transformation
\begin{equation*}
    A_l(z) = W_l z + b_l,
\end{equation*}
with weights $W_l \in \mathbb{R}^{d_l \times d_{l-1}}$ and biases $b_l \in
\mathbb{R}^{d_l}$. The activation function $\sigma: \mathbb{R} \to \mathbb{R}$
is applied componentwise.

The collection of all trainable parameters is denoted by
\begin{equation*}
\theta = \{ (W_l, b_l) \}_{l=1}^L,
\end{equation*}
and fully determines the function $\hat{f}_\theta$ for a fixed activation
function $\sigma$. The set of all such realizable functions for a given
architecture $\mathbf{d}$ and activation function $\sigma$ is denoted by
$\mathcal{F}_{\mathbf{d}, \sigma}$.

\vspace{0.5em}
\noindent The architectural parameters are interpreted as follows:
\begin{itemize}
    \item The \emph{depth} $L$ is the number of affine transformations.
    \item The number of \emph{hidden layers} is $L-1$.
    \item The \emph{width} $d_l$ specifies the number of neurons in layer $l$.
    \item The \emph{layer structure} is encoded in the tuple $\mathbf{d}$.
    \item The \emph{activation function} $\sigma$ usually introduces nonlinearity.
\end{itemize}
\end{definition}

As common in literature, the \emph{number of layers} is referenced by the number
of affine transformations $L$ and coincides with the term \emph{depth}. The
input layer is not counted as a layer in this context. In
Figure~\ref{fig:dnn-architecture}, a fully connected \ac{dnn} with constant
width is illustrated.

\begin{remark} \ \\
    For common \ac{dnn} architectures with $L$ layers, a constant number of
    neurons $m$ in each layer, input dimension $d_0=s$ and output dimension
    $d_L=1$ the layer structure follows:
    \begin{equation*}
        d=(d_0,d_1,...,d_{L-1},d_L) = (s,m,\dots,m,1)
    \end{equation*}
    Thus, for each affine transformation $l\in\{1,...,L\}$, one counts $d_{l-1}
    \cdot d_l$ weights and $d_l$ biases. Hence, the total number of trainable
    parameters (weights and biases) $\theta$ in the network is given by
    \begin{align*}
        N_\theta &= \sum_{l=1}^{L} (d_{l-1} \cdot d_l + d_l) \\
        &= (s \cdot m + m) + \sum_{l=2}^{L-1} (m \cdot m + m) + (m \cdot 1 + 1) \\
        &= (L-2) \cdot m^2 + (L+s) \cdot m + 1.
    \end{align*}
\end{remark}

\section{DNN Evaluation}
\label{sec:dnn-error}
% ------------------------------------------------------------------------------

The training of a \ac{dnn} involves supervised learning, where the goal is to
minimize the difference between the surrogate model $\hat{f}$ and the ground
truth function $f$. Supervised training of a \ac{dnn} therefore requires
minimizing the difference between the network output $\hat{f}_\theta(y)$ and a
ground truth function $f(y)$ on a given training set $\{y_i\}_{i=1}^N$.

We start by defining the loss function which is used to measure this difference
during the training:

\begin{definition}[Loss Function] \ \\
Let $f:[0,1]^s\to\mathbb{R}$ be a ground truth function defined on the hypercube
and $\hat{f}_\theta$ a respective surrogate with network architecture
$\mathbf{d}$, according parameters $\theta$ and activation function
$\boldsymbol{\sigma}$. Then, given the surrogate and a respective point set
$\mathcal{P}$, the \emph{loss function} is defined as
\begin{equation}
\mathcal{L}_{\mathcal{P}}(\theta) := \frac{1}{|\mathcal{P}|}\sum\limits_{y \in \mathcal{P}} |f(y) - \hat{f}_\theta(y)|^p\,,
\label{eq:loss_function}
\end{equation}
for some $1\leq p < \infty$.
\end{definition}

The point set $\mathcal{P}$ in the loss function is usually given by the training set. By defining the loss on the training data, the \emph{training error} $\mathcal{E}_T$ is defined:

\begin{definition}[Training Error] \ \\
Let $f:[0,1]^s \to \mathbb{R}$ be a ground truth function defined on the
hypercube and $\hat{f}_\theta$ a respective surrogate model with architecture
$\mathbf{d}$, according parameters $\theta$ and activation function
$\boldsymbol{\sigma}$. Then, given the matching training set $\{(y_i,
f(y_i))\}_{i=1}^N$ with $y_i \in [0,1]^s$ of the surrogate model, the
\emph{training error} is defined as
 \begin{equation*}
\mathcal{E}_\mathrm{T}(\theta) := \frac{1}{N} \sum_{i=1}^N |f(y_i) - \hat{f}_\theta(y_i)|.
 \end{equation*}
\end{definition}

The \emph{generalization error} $\mathcal{E}_\mathrm{G}$ is evaluating the
difference between the surrogate model $\hat{f}_\theta$ and the ground truth
function $f$ on the entire domain $[0,1]^s$.

\begin{definition}[Generalization Error] \ \\
Let $f:[0,1]^s\to\mathbb{R}$ be a ground truth function defined on the hypercube
and $\hat{f}_\theta$ a respective surrogate with nework architecture
$\mathbf{d}$, according parameters $\theta$ and activation function
$\boldsymbol{\sigma}$. Then, given the surrogate, the \emph{generalization
error} is defined as:
\begin{equation*}
\mathcal{E}_\mathrm{G}(\theta) := \int\limits_{[0,1]^s} |f(y) - \hat{f}_\theta(y)| \, \mathrm{d}y.
\end{equation*}
\end{definition}

\begin{remark} \ \\
    Whereas the training error $\mathcal{E}_\mathrm{T}$ is numerically
    straightforward to compute, the generalization error
    $\mathcal{E}_\mathrm{G}$ is often not directly computable, as the ground
    truth function $f$ could not be known besides the training data or is costly
    to compute.
\end{remark}

\begin{definition}[Generalization Gap] \ \\
    The \emph{generalization gap} for a given \ac{dnn} architecture
    $\mathbf{d}$, according parameters $\theta$ and activation function
    $\boldsymbol{\sigma}$ is the difference between the generalization and
    training error:
    \begin{equation*}
        |\mathcal{E}_\mathrm{G}(\theta) - \mathcal{E}_\mathrm{T}(\theta)|.
    \end{equation*}
\end{definition}

\section{Optimization and Training Procedure}
\label{sec:dnn-training}
% ------------------------------------------------------------------------------
In practice, an initial surrogate \ac{dnn} $\hat{f}_\theta$ is generated with
initial parameters $\theta$. In supervised learning, the goal is to minimize the
\emph{regularized loss function} with respect to the parameters $\theta$ by
adapting the weights of the network during the training process. Regularization
is introduced to counteract \emph{overfitting}, i.e. the tendency of a highly
flexible network to reproduce noise of the training data instead of the
underlying function. The regularization term penalizes certain properties of the
parameters, thereby enforcing additional structure on the solution. For
instance, an $L^2$ penalty discourages large weight magnitudes and effectively
promotes smoother functions.

Thus, one aims to find a solution to the following minimization problem:
\begin{equation*}
    \theta^* = \arg\min_{\theta} \mathcal{J}_\mathcal{P}(\theta),
    \quad \text{with } 
    \mathcal{J}(\theta) = \mathcal{L}_\mathcal{P}(\theta) + \lambda \cdot R(\theta),
\end{equation*}
where $\mathcal{P}$ is the training set, $\mathcal{L}_\mathcal{P}$ is the
empirical loss (e.g. mean squared error), $R(\theta)$ is a regularization
functional and $\lambda > 0$ is the regularization parameter balancing fidelity
to the training data and the strength of the imposed constraint.

The training is carried out on a given \emph{training dataset} consisting of
$N$ input--output pairs ${(x_n, f(x_n))}_{n=1}^N$, where $N$ is called the
\emph{training set size}. Since the dataset is usually too large to be processed
at once, it is divided into smaller subsets of fixed cardinality $B$, called
\emph{mini-batches}. The number $B$ is referred to as the \emph{batch size} and
determines how many samples are processed simultaneously in each training step.

The training procedure is organized into multiple passes over the dataset,
called \emph{epochs}. At the beginning of each epoch $e \in {1, \dots, E}$, the
dataset is typically shuffled and then partitioned into mini-batches
\begin{equation*}
(\mathcal{B}^e_k)_{k=1}^{K}, \quad
\text{with } K = \left\lceil \tfrac{N}{B} \right\rceil,
\end{equation*}
such that the union of all mini-batches corresponds to the full training set.
This ensures that in each epoch, the entire dataset is used for training.

For each mini-batch $\mathcal{B}^e_k$, the algorithm executes the following
steps: the current surrogate network $\hat{f}_{\theta}$ is applied to all
samples in $\mathcal{B}^e_k$ in a forward pass and the corresponding
\emph{regularized loss function} $\mathcal{J}_{\mathcal{B}_k^e}$ is evaluated
with respect to these samples. The gradient of the loss with respect to the
parameters, $\nabla_\theta \mathcal{J}_{\mathcal{B}_k^e}$, is then computed by
backpropagation. Finally, the parameters $\theta$ are updated according to the
chosen optimization algorithm $\mathrm{Opt}$, e.g. SGD, Adam or Lion as
introduced in the followup subsections.

This procedure is repeated over all mini-batches of all epochs or until a
stopping criterion is met, resulting in the final trained network
$\hat{f}_{\theta^*} = \hat{f}_{\theta_E}$.

Training deep neural networks requires the minimization of a loss function
$\mathcal{J}(\theta)$ with respect to the network parameters $\theta$. As
computing the exact minimum is intractable in practice, iterative first-order
methods are typically employed. In this section, two widely used optimization
algorithms are briefly introduced: \acl{sgd} and Adam. Since the primary focus
of this thesis lies on sampling strategies rather than on optimization methods,
the discussion remains concise; for a more detailed treatment, see
\cite[Chapter~8]{goodfellow2016deep}.

\subsection{Stochastic Gradient Descent (SGD)}

The most basic algorithm for training neural networks is Gradient Descent. As defined above, given the entire dataset $\mathcal{P}$, the Gradient Descent update is given by
\begin{equation*}
    \theta_{k+1} = \theta_k - \eta \nabla_\theta \mathcal{J}_{\mathcal{P}}(\theta_k), 
\end{equation*}
where $\eta>0$ is the learning rate.

Computing this full gradient at every iteration is costly for large
datasets. The idea of \emph{\acl{sgd}} is to approximate it by using a randomly
chosen mini-batch $\mathcal{B}_k \subset \mathcal{P}$ at each iteration $k$:
$$
    g_k = \nabla_\theta \mathcal{J}_{\mathcal{B}_k}(\theta_{k-1}).
$$
The parameter update is then given by
$$
    \theta_k = \theta_{k-1} - \eta g_k,
$$


The crucial point is that $g_k$ is an \emph{exact gradient} on the mini-batch,
but only an \emph{unbiased stochastic estimator} of the full gradient. This
randomness in the update direction explains why the method is called
\emph{stochastic gradient descent}. The gradient itself is computed efficiently
by backpropagation. Further details on backpropagation can be found in
\cite[Section~6.5]{goodfellow2016deep}.

A widely used extension of \ac{sgd} is \emph{momentum}. Instead of following the
raw stochastic gradient at each step, the algorithm maintains a moving average
of past gradients, the second momentum
$$
    v_k = \beta_2 v_{k-1} + (1-\beta_2) g_k^2,
$$
with $\beta_2 \in [0,1)$. The updated gradient step is then scaled by this
second momentum:

$$
    \theta_k = \theta_{k-1} - \frac{\eta}{v_k}g_k\,.
$$

Intuitively, momentum accelerates progress along consistent descent directions
and damps oscillations in directions of high curvature, much like inertia in
physical systems.

\subsection{Adam Optimizer}

The Adam optimizer~\cite[Section~8.5]{goodfellow2016deep} extends the usage of
momentum. At each iteration, Adam maintains exponential moving averages of both
the gradient (first moment) and its elementwise square (second moment):
\begin{align*}
    m_k &= \beta_1 m_{k-1} + (1-\beta_1) g_k, \\
    v_k &= \beta_2 v_{k-1} + (1-\beta_2) g_k^2,
\end{align*}
where $\beta_1,\beta_2\in[0,1)$ are decay parameters.

The algorithm also includes bias-correction terms to account for the
initialization and is given in Algorithm~\ref{alg:adam}.

\begin{algorithm}[H]
\caption{Adam Optimizer}
\label{alg:adam}
\begin{algorithmic}[1]
\Require initial parameters $\theta_0$; learning rate $\eta>0$; decay rates
$\beta_1,\beta_2\in[0,1)$; small $\varepsilon>0$ \State $m_0 \gets 0$ \State
$v_0 \gets 0$ \For{$k=1,2,\dots$} \State sample mini-batch $\mathcal{B}_k
\subset \mathcal{P}$ \State $g_k \gets \nabla_\theta
\mathcal{J}_{\mathcal{B}_k}(\theta_{k-1})$ \State $m_k \gets \beta_1 m_{k-1} +
(1-\beta_1) g_k$ \Comment{first momentum} \State $v_k \gets \beta_2 v_{k-1} +
(1-\beta_2) g_k^{\;2}$ \Comment{second momentum} \State $\hat m_k \gets
\dfrac{m_k}{1-\beta_1^{\,k}}$ \Comment{bias correction} \State $\hat v_k \gets
\dfrac{v_k}{1-\beta_2^{\,k}}$ \Comment{bias correction} \State $\theta_k \gets
\theta_{k-1} - \eta \; \dfrac{\hat m_k}{\sqrt{\hat v_k}+\varepsilon}$ \EndFor
\end{algorithmic}
\end{algorithm}

Overall, Adam combines the stabilizing effect of momentum with per-parameter
adaptive learning rates, which makes it one of the most robust and widely used
optimizers in modern deep learning applications.
